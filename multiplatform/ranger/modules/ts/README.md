# TypeScript Module Snapshot

This directory contains TypeScript outputs for the NG parser.

Contents:
- ng/token_detector_ng.ts: native Ranger TypeScript output (`-typescript` flag, currently experimental)
- ng/index.ts: stable runtime facade over JS output
- ng/index.d.ts: typed declarations for the stable facade

Usage:

```ts
import { parseTokens } from './ng/index.js';

const input = '[2026-02-23] ## Uinti\nSwim; 400m\n> Split 100m 2:45/100m';
const rows = parseTokens(input);
console.log(rows.length);
```

## Parsing example (harness-style Swim + nested Split/Recovery)

Create `your_app.ts`:

```ts
import { Parser, StandardDetectors, TokenSlice } from './ng/token_detector_ng.js';

function parseTokens(input: string): TokenSlice[] {
  const parser = new Parser(input, StandardDetectors.create());
  parser.start();
  return parser.getResults();
}

function firstChild(token: TokenSlice, tag: string): TokenSlice | undefined {
  for (let i = 0; i < token.childCount(); i++) {
    const child = token.getChild(i);
    if (child.tag === tag) return child;
  }
  return undefined;
}

const workout = `[2026-02-23] ## Uinti
Swim; 400m
> Split 100m 2:45/100m
>> Recovery 1min`;

// Parsitaan koko blokki yhdellä kertaa, kuten harnessissa.
const rows = parseTokens(workout);
console.log('Row token count:', rows.length);

// 0) Swim; 400m
if (rows.length > 0) {
  const swimRB = firstChild(rows[0], 'repeat-block');
  if (swimRB && swimRB.hasRepeatBlockValue()) {
    const swim = swimRB.getAsRepeatBlockValue();
    console.log('Swim part count =', swim.parts.length);
  }
}

// 1) > Split 100m 2:45/100m
if (rows.length > 1) {
  const splitRow = rows[1];

  if (splitRow.childCount() > 0) {
    const levelToken = splitRow.getChild(0);
    if (levelToken.hasDetailsLevelValue()) {
      const level = levelToken.getAsDetailsLevelValue();
      console.log('Split details level =', level.level, 'marker=', level.marker);
    }
  }

  if (splitRow.childCount() > 2) {
    const splitDistanceToken = splitRow.getChild(2);
    if (splitDistanceToken.hasRepeatBlockValue()) {
      const d = splitDistanceToken.getAsRepeatBlockValue();
      console.log('Split repeat block parts =', d.parts.length);
    }
  }

  if (splitRow.childCount() > 3) {
    const speedToken = splitRow.getChild(3);
    if (speedToken.childCount() >= 3) {
      const tvToken = speedToken.getChild(0);
      const dvToken = speedToken.getChild(2);
      if (tvToken.hasTimeValueValue() && dvToken.hasDistanceValue()) {
        const tv = tvToken.getAsTimeValueValue();
        const dv = dvToken.getAsDistanceValue();
        console.log(`Split pace value = ${tv.minutes}:${String(tv.seconds).padStart(2,'0')}/${dv.value}${dv.unit}`);
      }
    }
  }
}

// 2) >> Recovery 1min
if (rows.length > 2) {
  const recoveryRow = rows[2];

  if (recoveryRow.childCount() > 0) {
    const levelToken = recoveryRow.getChild(0);
    if (levelToken.hasDetailsLevelValue()) {
      const level = levelToken.getAsDetailsLevelValue();
      console.log('Recovery details level =', level.level, 'marker=', level.marker);
    }
  }

  if (recoveryRow.childCount() > 1) {
    const recToken = recoveryRow.getChild(1);
    if (recToken.hasRecoveryValue()) {
      const rv = recToken.getAsRecoveryValue();
      console.log('Recovery label =', rv.label);
    }
    if (recToken.childCount() > 1) {
      const rtToken = recToken.getChild(1);
      if (rtToken.hasRecoveryTimeValue()) {
        const rt = rtToken.getAsRecoveryTimeValue();
        console.log('Recovery time value =', rt.value + rt.unit);
      }
    }
  }
}

/*
Expected style of output:
Row token count: 3
Swim part count = 1
Split details level = 1 marker= >
Split repeat block parts = 1
Split pace value = 2:45/100m
Recovery details level = 2 marker= >>
Recovery label = Recovery
Recovery time value = 1min
*/
```

Compile and run:

```bash
npx tsc --esModuleInterop --module nodenext --moduleResolution nodenext \
  multiplatform/ranger/modules/ts/ng/token_detector_ng.ts \
  your_app.ts \
  --outDir dist_app

node dist_app/your_app.js
```

Note:
- `token_detector_ng.ts` is generated by Ranger compiler using `-typescript`.
- In current Ranger version this native TS output may fail strict `tsc` checks for complex sources.
- Prefer `index.ts` + `index.d.ts` in consuming projects until native TS target matures.
